cmake_minimum_required(VERSION 3.21)
project(TahoePhotoBooth VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.9 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Multimedia ShaderTools DBus)
qt_standard_project_setup()
add_library(booth-desktop-integration STATIC src/cameradiscovery.cpp src/cameradiscovery.h src/nautilussavedialog.cpp src/nautilussavedialog.h)
target_link_libraries(booth-desktop-integration PUBLIC Qt6::Gui PRIVATE Qt6::Multimedia Qt6::DBus)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Qt6 REQUIRED COMPONENTS MultimediaPrivate GuiPrivate)
  target_link_libraries(booth-desktop-integration PRIVATE Qt6::MultimediaPrivate Qt6::GuiPrivate)
endif()
file(GLOB_RECURSE PHOTOBOOTH_WINDOW_CONTROLS CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "assets/window-controls/*.png")
qt_add_executable(tahoe-photobooth src/main.cpp src/booth.cpp src/booth.h src/effects.cpp src/effects.h src/frameview.cpp src/frameview.h)
qt_add_resources(tahoe-photobooth "ui" PREFIX "/" FILES qml/Main.qml qml/ActionButton.qml qml/ButtonSurface.qml qml/Glyph.qml qml/SettingsGlass.qml qml/DockToolTip.qml qml/ContinuousPanel.qml packaging/org.macshell.PhotoBooth.svg ${PHOTOBOOTH_WINDOW_CONTROLS})
qt_add_shaders(tahoe-photobooth "settings_shaders" PREFIX "/" FILES shaders/settings-glass.frag shaders/settings-content.frag)
target_link_libraries(tahoe-photobooth PRIVATE booth-desktop-integration Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia)
target_compile_options(tahoe-photobooth PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS tahoe-photobooth RUNTIME DESTINATION bin)
install(FILES packaging/org.macshell.PhotoBooth.svg DESTINATION share/icons/hicolor/512x512/apps)
# Change the desktop icon URL when the image changes, invalidating Qt image caches.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/packaging/org.macshell.PhotoBooth.svg")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/packaging/org.macshell.PhotoBooth.svg" PHOTOBOOTH_ICON_HASH)
string(SUBSTRING "${PHOTOBOOTH_ICON_HASH}" 0 12 PHOTOBOOTH_ICON_HASH)
set(PHOTOBOOTH_DESKTOP_ICON "org.macshell.PhotoBooth-${PHOTOBOOTH_ICON_HASH}.svg")
install(FILES packaging/org.macshell.PhotoBooth.svg DESTINATION share/icons/hicolor/512x512/apps RENAME "${PHOTOBOOTH_DESKTOP_ICON}")
configure_file(packaging/org.macshell.PhotoBooth.desktop.in org.macshell.PhotoBooth.desktop @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/org.macshell.PhotoBooth.desktop" DESTINATION share/applications)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(booth-native-export-tests tests/native_export_test.cpp)
  target_link_libraries(booth-native-export-tests PRIVATE booth-desktop-integration Qt6::Test Qt6::DBus)
  add_test(NAME native-nautilus-export COMMAND booth-native-export-tests)
  qt_add_executable(booth-tests tests/booth_test.cpp src/booth.cpp src/booth.h src/effects.cpp src/effects.h)
  target_link_libraries(booth-tests PRIVATE booth-desktop-integration Qt6::Quick Qt6::Multimedia Qt6::Test)
  add_test(NAME capture-effects-library-recording COMMAND booth-tests)
  set_tests_properties(capture-effects-library-recording PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_MEDIA_BACKEND=ffmpeg" TIMEOUT 45)
  qt_add_executable(booth-ui-tests tests/ui_test.cpp src/booth.cpp src/booth.h src/effects.cpp src/effects.h src/frameview.cpp src/frameview.h)
  qt_add_resources(booth-ui-tests "testui" PREFIX "/" FILES qml/Main.qml qml/ActionButton.qml qml/ButtonSurface.qml qml/Glyph.qml qml/SettingsGlass.qml qml/DockToolTip.qml qml/ContinuousPanel.qml ${PHOTOBOOTH_WINDOW_CONTROLS})
  qt_add_shaders(booth-ui-tests "test_settings_shaders" PREFIX "/" FILES shaders/settings-glass.frag shaders/settings-content.frag)
  target_link_libraries(booth-ui-tests PRIVATE booth-desktop-integration Qt6::Quick Qt6::QuickControls2 Qt6::Multimedia Qt6::Test)
  add_test(NAME desktop-ui-interactions COMMAND booth-ui-tests)
  set_tests_properties(desktop-ui-interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_MEDIA_BACKEND=ffmpeg" TIMEOUT 35)
endif()
