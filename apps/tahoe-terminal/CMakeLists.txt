cmake_minimum_required(VERSION 3.20)
project(TahoeTerminal VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET webkit2gtk-4.1)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-glib-1.0)
add_library(tahoe-pty src/pty.cpp)
target_include_directories(tahoe-pty PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_link_libraries(tahoe-pty PUBLIC PkgConfig::GTK util)
target_compile_options(tahoe-pty PRIVATE -Wall -Wextra -Wpedantic)
add_executable(tahoe-terminal src/main.cpp)
target_link_libraries(tahoe-terminal PRIVATE tahoe-pty PkgConfig::WEBKIT PkgConfig::JSON)
target_compile_options(tahoe-terminal PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(tahoe-terminal PRIVATE TAHOE_SOURCE_WEB="${CMAKE_CURRENT_SOURCE_DIR}/web")
install(TARGETS tahoe-terminal RUNTIME DESTINATION bin)
install(DIRECTORY web/ DESTINATION share/tahoe-terminal/web)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/packaging/tahoe-terminal.desktop" desktop_entry)
string(REPLACE "\nExec=tahoe-terminal" "\nExec=\"${CMAKE_INSTALL_PREFIX}/bin/tahoe-terminal\"" desktop_entry "${desktop_entry}")
string(REPLACE "\nTryExec=tahoe-terminal" "\nTryExec=${CMAKE_INSTALL_PREFIX}/bin/tahoe-terminal" desktop_entry "${desktop_entry}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/tahoe-terminal.desktop" "${desktop_entry}")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/tahoe-terminal.desktop" DESTINATION share/applications)
install(FILES packaging/tahoe-terminal.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES README.md LICENSE DESTINATION share/doc/tahoe-terminal)
include(CTest)
if(BUILD_TESTING)
  add_executable(pty-test tests/pty_test.cpp)
  target_link_libraries(pty-test PRIVATE tahoe-pty)
  add_test(NAME real-pty-session COMMAND pty-test)
  set_tests_properties(real-pty-session PROPERTIES TIMEOUT 20)
  add_executable(flow-test tests/flow_test.cpp)
  target_link_libraries(flow-test PRIVATE tahoe-pty)
  add_test(NAME stream-backpressure-and-interrupt COMMAND flow-test)
  set_tests_properties(stream-backpressure-and-interrupt PROPERTIES TIMEOUT 15)
endif()
